
CREATE SEQUENCE tb_vender_password_history_id_seq;
CREATE TABLE IF NOT EXISTS tb_vender_password_history
(
    id numeric(10,0) NOT NULL DEFAULT nextval('tb_vender_password_history_id_seq'),
    account character varying(20) COLLATE pg_catalog."default",
    pw character varying(150) COLLATE pg_catalog."default",
    createtime timestamp without time zone,
    locked_reason smallint
);
 
COMMENT ON COLUMN tb_vender_password_history.createtime     IS '廠商密碼變更時間';
COMMENT ON COLUMN tb_vender_password_history.account     IS '廠商帳號';
COMMENT ON COLUMN tb_vender_password_history.pw     IS '廠商密碼';
COMMENT ON COLUMN tb_vender_password_history.locked_reason     IS '廠商鎖定原因';
